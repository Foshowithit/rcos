**Engine design:**
- Takes argv: field_map_path, records_path, out_path
- field_map_path: a JSON file describing how to extract canonical fields from source records (schema adapter)
- records_path: JSON file with records from source A (e.g., orders) and source B (e.g., shipments)
- Produces reconciled output

**Field map format:**
```json
{
  "match_keys": {"order_id": ["source_a.order_id", "source_b.order_id"]},
  "field_mappings": {
    "quantity": {"a": "qty_ordered", "b": "qty_received"},
    "status": {"a": "order_status", "b": "shipment_status"}
  }
}
```

**Classification:**
- equal: all mapped fields match
- missing_in_b: key only in A
- missing_in_a: key only in B
- conflicts: key in both but fields don't match
- unreconcilable: parse errors or missing match keys

```python
#!/usr/bin/env python3
import json, sys

def get_field(rec, path):
    parts = path.split('.')
    cur = rec
    for p in parts:
        if isinstance(cur, dict) and p in cur:
            cur = cur[p]
        else:
            return None
    return cur

def normalize(val):
    if val is None:
        return None
    if isinstance(val, (int, float)):
        return val
    return str(val).strip().lower()

def main():
    field_map_path, records_path, out_path = sys.argv[1], sys.argv[2], sys.argv[3]
    with open(field_map_path) as f:
        fmap = json.load(f)
    with open(records_path) as f:
        data = json.load(f)
    
    a_records = data.get('a', [])
    b_records = data.get('b', [])
    match_key_paths = fmap['match_keys']
    field_mappings = fmap['field_mappings']
    
    def build_index(records, key_paths):
        idx = {}
        for rec in records:
            key_vals = tuple(get_field(rec, kp) for kp in key_paths)
            if all(v is not None for v in key_vals):
                idx[key_vals] = rec
            else:
                idx[key_vals] = rec  # unreconcilable
        return idx
    
    a_idx = {}
    a_bad = []
    for rec in a_records:
        key_vals = tuple(get_field(rec, kp) for kp in match_key_paths)
        if all(v is not None for v in key_vals):
            a_idx[key_vals] = rec
        else:
            a_bad.append(rec)
    
    b_idx = {}
    b_bad = []
    for rec in b_records:
        key_vals = tuple(get_field(rec, kp) for kp in match_key_paths)
        if all(v is not None for v in key_vals):
            b_idx[key_vals] = rec
        else:
            b_bad.append(rec)
    
    matched = []
    conflicts = []
    missing_in_b = []
    missing_in_a = []
    
    for key, a_rec in a_idx.items():
        if key not in b_idx:
            missing_in_b.append({'key': list(key), 'record_a': a_rec})
            continue
        b_rec = b_idx[key]
        field_results = {}
        all_equal = True
        for canonical, paths in field_mappings.items():
            a_val = normalize(get_field(a_rec, paths.get('a')))
            b_val = normalize(get_field(b_rec, paths.get('b')))
            field_results[canonical] = {'a': a_val, 'b': b_val, 'equal': a_val == b_val}
            if a_val != b_val:
                all_equal = False
        if all_equal:
            matched.append({'key': list(key), 'fields': field_results})
        else:
            conflicts.append({'key': list(key), 'record_a': a_rec, 'record_b': b_rec, 'fields': field_results})
    
    for key, b_rec in b_idx.items():
        if key not in a_idx:
            missing_in_a.append({'key': list(key), 'record_b': b_rec})
    
    unreconcilable = [{'source': 'a', 'record': r} for r in a_bad] + [{'source': 'b', 'record': r} for r in b_bad]
    
    result = {
        'matched': matched,
        'missing_in_b': missing_in_b,
        'missing_in_a': missing_in_a,
        'conflicts': conflicts,
        'unreconcilable': unreconcilable,
        'stats': {
            'total_a': len(a_records),
            'total_b': len(b_records),
            'matched': len(matched),
            'conflicts': len(conflicts),
            'missing_in_b': len(missing_in_b),
            'missing_in_a': len(missing_in_a),
            'unreconcilable': len(unreconcilable)
        }
    }
    
    with open(out_path, 'w') as f:
        json.dump(result, f, indent=2)

if __name__ == '__main__':
    main()
```
