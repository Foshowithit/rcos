#!/usr/bin/env python3
import sys
import os
import json
import subprocess

def main(src_dir, dst_path):
    task_dir = os.path.abspath(src_dir)
    counter_dir = os.path.join(task_dir, "counter")
    os.makedirs(counter_dir, exist_ok=True)

    fetch_script = os.path.join(task_dir, "fetch.py")

        page_data = None

                result = subprocess.run(
