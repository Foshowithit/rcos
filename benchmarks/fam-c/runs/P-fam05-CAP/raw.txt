#!/usr/bin/env python3
import sys, json, hashlib

def sha256_hex(data):
    if isinstance(data, str):
        data = data.encode('utf-8')
    return hashlib.sha256(data).hexdigest()

def main():
    try:
        if len(sys.argv) != 4:
            print("usage: engine.py field_map records out", file=sys.stderr)
            return 1
        
        field_map_path, records_path, out_path = sys.argv[1], sys.argv[2], sys.argv[3]
        
        with open(field_map_path, 'r') as f:
            field_map = json.load(f)
        
        with open(records_path, 'r') as f:
            records = json.load(f)
        
        ok = []
        bad = []
        unverified = []
        
        for path, record in records.items():
            if not isinstance(record, dict):
                bad.append({"path": path, "reason": f"not a dict: {type(record).__name__}"})
                continue
            
            spec = field_map.get(path)
            if spec is None:
                # No spec means we can't verify, or could be unverified
                unverified.append(path)
                continue
            
            entry_ok = True
            reasons = []
            
            for field, value in record.items():
                expected = spec.get(field, {})
                
                # Size check
                exp_size = expected.get('size')
                if exp_size is not None:
                    actual_size = len(value) if isinstance(value, (str, bytes)) else len(str(value).encode('utf-8'))
                    if actual_size != exp_size:
                        entry_ok = False
                        reasons.append(f"{field}: size {actual_size} != expected {exp_size}")
                
                # sha256 check
                exp_sha = expected.get('sha256')
                if exp_sha is not None:
                    actual_sha = sha256_hex(value)
                    if actual_sha != exp_sha:
                        entry_ok = False
                        reasons.append(f"{field}: sha256 mismatch")
            
            # Check required fields present
            for field, expected in spec.items():
                if expected.get('required', False) and field not in record:
                    entry_ok = False
                    reasons.append(f"missing required field: {field}")
            
            if not entry_ok:
                bad.append({"path": path, "reason": "; ".join(reasons)})
            else:
                # Check if any field had no expected values -> unverified
                has_unverified = False
                for field, value in record.items():
                    expected = spec.get(field, {})
                    if 'size' not in expected and 'sha256' not in expected:
                        has_unverified = True
                        break
                if has_unverified:
                    unverified.append(path)
                else:
                    ok.append(path)
        
        with open(out_path, 'w') as f:
            json.dump({"ok": ok, "bad": bad, "unverified": unverified}, f, indent=2)
        
        return 0
    except Exception as e:
        print(f"crash: {e}", file=sys.stderr)
        return 1

if __name__ == '__main__':
    sys.exit(main())
