```python
import hashlib
import os
import sys
import json

def solve(manifest_path, files_dir, out_path):
    with open(manifest_path, 'r') as f:
        lines = f.readlines()
    
    files = []
    sizes = []
    shas = []
    for line in lines:
        line = line.strip()
        if line.startswith('FILES='):
            files = line[len('FILES='):].split(',')
        elif line.startswith('SIZES='):
            sizes = [int(s) for s in line[len('SIZES='):].split(',')]
        elif line.startswith('SHAS='):
            shas = line[len('SHAS='):].split(',')
    
    ok = []
    bad = []
    unverified = []
    
    for i, fname in enumerate(files):
        fpath = os.path.join(files_dir, fname)
        if not os.path.exists(fpath):
            unverified.append(fname)
            continue
        
        actual_size = os.path.getsize(fpath)
        expected_size = sizes[i]
        expected_sha = shas[i]
        
        with open(fpath, 'rb') as f:
            data = f.read()
        actual_sha = hashlib.sha256(data).hexdigest()
        
        if actual_size != expected_size:
            bad.append({"path": fname, "reason": f"size mismatch: expected {expected_size}, got {actual_size}"})
        elif actual_sha != expected_sha:
            bad.append({"path": fname, "reason": f"sha mismatch: expected {expected_sha}, got {actual_sha}"})
        else:
            ok.append(fname)
    
    result = {"ok": ok, "bad": bad, "unverified": unverified}
    with open(out_path, 'w') as f:
        json.dump(result, f)

if __name__ == '__main__':
    solve(sys.argv[1], sys.argv[2], sys.argv[3])
```
