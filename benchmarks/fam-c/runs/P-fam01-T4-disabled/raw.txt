import csv
import json
import os
from decimal import Decimal

def solve(src_dir, dst_path):
    csv_path = os.path.join(src_dir, 'input.csv')
    records = []
    with open(csv_path, 'r', newline='') as f:
        reader = csv.DictReader(f)
        for row in reader:
            # Skip rows without an id (empty rows or summary rows)
            if not row.get('id'):
                continue
            # Skip the SUMMARY row
            if row['id'] == 'SUMMARY':
                continue
            amount_str = row.get('amount_usd', '') or '0'
            # Convert amount to cents
            amount_cents = int((Decimal(amount_str) * 100).to_integral_value())
            tags_str = row.get('tags', '') or ''
            tags = [t.strip() for t in tags_str.split(',') if t.strip()] if tags_str else []
            records.append({
                'id': row['id'],
                'name': row.get('name', '') or '',
                'amount_cents': amount_cents,
                'tags': tags,
            })
    
    with open(dst_path, 'w') as f:
        json.dump(records, f, indent=2)
