```python
import json
import os
import sys

def solve(src_dir, dst_path):
    with open(os.path.join(src_dir, 'input.env'), 'r') as f:
        content = f.read()
    
    records = []
    current_id = None
    current = {}
    
    for line in content.splitlines():
        line = line.rstrip('\n').rstrip('\r')
        stripped = line.strip()
        if not stripped or stripped.startswith('#'):
            continue
        if stripped.startswith('[') and stripped.endswith(']'):
            if current_id is not None:
                records.append(finalize(current_id, current))
            current_id = stripped[1:-1]
            current = {}
        elif '=' in stripped:
            key, _, value = stripped.partition('=')
            current[key.strip()] = value.strip()
    
    if current_id is not None:
        records.append(finalize(current_id, current))
    
    with open(dst_path, 'w') as f:
        json.dump(records, f, indent=2)

def finalize(rid, data):
    usd_str = data.get('usd', '0')
    amount_cents = int(round(float(usd_str) * 100))
    tags_str = data.get('tags', '')
    tags = [t for t in tags_str.split('|') if t]
    return {
        'id': rid,
        'name': data.get('name', ''),
        'amount_cents': amount_cents,
        'tags': tags,
    }

if __name__ == '__main__':
    solve(sys.argv[1], sys.argv[2])
```
