import sys, json

def parse_usd_to_cents(s):
    if s is None:
        return 0
    s = str(s).strip()
    neg = s.startswith('-') or s.startswith('(')
    if s.startswith('('):
        s = s.strip(')')
    s = s.lstrip('$€£¥').strip()
    s = s.replace(',', '')
    if neg and not s.startswith('-'):

    if neg:
        s = s[1:]
    if '.' in s:
        a, b = s.split('.', 1)
        b = (b + '00')[:2]
    else:
        a, b = s, '00'
    cents = (int(a) if a else 0) * 100 + int(b)
    return -cents if neg else cents

def pick(rec, keys):
