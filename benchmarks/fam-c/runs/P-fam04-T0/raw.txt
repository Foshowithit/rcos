    visited = set()

import json
import os
import sys
from collections import deque, defaultdict

def main():
    src_dir = sys.argv[1]
    dst_path = sys.argv[2]
    
    with open(os.path.join(src_dir, 'graph.json'), 'r') as f:
        graph = json.load(f)
    
    nodes = graph['nodes']
    edges = graph['edges']
    
    adj = defaultdict(list)
    in_degree = {node: 0 for node in nodes}
    
    for u, v in edges:
        adj[u].append(v)
        in_degree[v] = in_degree.get(v, 0) + 1
        # Ensure all nodes appear in adj even if no outgoing edges
        if u not in in_degree:
            in_degree[u] = 0
    
    # Kahn's algorithm
    queue = deque([node for node in nodes if in_degree[node] == 0])
    order = []
    
    while queue:
        u = queue.popleft()
        order.append(u)
        for v in adj[u]:
            in_degree[v] -= 1
            if in_degree[v] == 0:
                queue.append(v)
    
    if len(order) == len(nodes):
        result = {"valid": True, "order": order}
    else:
        # Find a cycle using DFS
        cycle = find_cycle(adj, nodes)
        result = {"valid": False, "cycle": cycle}
    
    with open(dst_path, 'w') as f:
        json.dump(result, f)

def find_cycle(adj, all_nodes):

    rec_stack = set()
